cmake_minimum_required(VERSION 3.24)
project(aiInterview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_DESKTOP_CLIENT "Build the Qt desktop client" ON)
option(BUILD_BACKEND "Build the business gRPC backend" OFF)
option(BUILD_CPP_GATEWAY "Build the protocol-only C++ gateway" OFF)
option(BUILD_PROJECT_TESTS "Build Qt smoke tests" OFF)
add_subdirectory(proto)
if(BUILD_DESKTOP_CLIENT)
    add_subdirectory(client)
endif()
if(BUILD_BACKEND)
    include(CTest)
    add_subdirectory(server)
elseif(BUILD_CPP_GATEWAY)
    include(CTest)
    add_subdirectory(server/gateway)
endif()
