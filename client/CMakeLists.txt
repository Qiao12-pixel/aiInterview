cmake_minimum_required(VERSION 3.24)
project(aiInterviewClient LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_PROJECT_TESTS "Build Qt client smoke tests" OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../proto" "${CMAKE_CURRENT_BINARY_DIR}/protocol")
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
add_library(interview_client_transport src/transport/interview_client.cpp)
target_include_directories(interview_client_transport PUBLIC src)
target_link_libraries(interview_client_transport PUBLIC interview_protocol)
set(CLIENT_UI_SOURCES
    src/ui/MainWindow.h src/ui/config_dialog.h
    src/ui/MainWindow.cpp src/ui/window_navigation.cpp
    src/ui/window_layout.cpp src/ui/session_actions.cpp
    src/ui/answer_actions.cpp src/ui/report_actions.cpp
    src/ui/config_dialog.cpp src/ui/config_dialog_layout.cpp)
add_executable(aiInterview src/main_qt.cpp ${CLIENT_UI_SOURCES})
target_sources(aiInterview PRIVATE src/account/account_dialog.h src/account/account_dialog.cpp)
set_target_properties(aiInterview PROPERTIES AUTOMOC ON AUTOUIC ON)
target_include_directories(aiInterview PRIVATE src)
target_link_libraries(aiInterview PRIVATE interview_client_transport Qt6::Core Qt6::Widgets Qt6::Network)
if(BUILD_PROJECT_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(client_ui_smoke tests/ui_smoke.cpp ${CLIENT_UI_SOURCES})
    set_target_properties(client_ui_smoke PROPERTIES AUTOMOC ON AUTOUIC ON)
    target_link_libraries(client_ui_smoke PRIVATE interview_client_transport Qt6::Widgets Qt6::Test)
    add_executable(client_account_smoke tests/account_smoke.cpp src/account/account_dialog.h src/account/account_dialog.cpp)
    set_target_properties(client_account_smoke PROPERTIES AUTOMOC ON)
    target_link_libraries(client_account_smoke PRIVATE interview_client_transport Qt6::Widgets Qt6::Network)
    add_executable(client_rpc_smoke tests/rpc_smoke.cpp)
    target_link_libraries(client_rpc_smoke PRIVATE interview_client_transport)
endif()
# 聚合构建时保留根构建目录的 Qt 启动与测试产物路径。
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    foreach(target aiInterview client_ui_smoke client_account_smoke client_rpc_smoke)
        if(TARGET ${target})
            set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/..")
        endif()
    endforeach()
endif()
