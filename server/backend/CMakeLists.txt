cmake_minimum_required(VERSION 3.24)
project(aiInterviewBackend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../proto" "${CMAKE_CURRENT_BINARY_DIR}/protocol")
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-cpp)
pkg_check_modules(MYSQL REQUIRED IMPORTED_TARGET mysqlclient)
pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis)
find_package(OpenSSL REQUIRED)
add_subdirectory(src/infrastructure/database)
# 应用层只依赖领域与端口；具体 HTTP/PDF 实现不反向进入业务库。
add_library(interview_domain INTERFACE)
target_include_directories(interview_domain INTERFACE src)
target_link_libraries(interview_domain INTERFACE nlohmann_json::nlohmann_json)
add_library(interview_json src/common/json_output.cpp src/common/base64.cpp)
target_link_libraries(interview_json PUBLIC interview_domain)
add_library(interview_application src/application/engine.cpp src/application/answer.cpp src/application/report.cpp)
target_link_libraries(interview_application PUBLIC interview_domain interview_json)
add_library(interview_infrastructure src/infrastructure/llm.cpp src/infrastructure/llm/contract.cpp src/infrastructure/llm/mock.cpp src/infrastructure/llm/config.cpp src/infrastructure/llm/http.cpp src/infrastructure/llm/stream.cpp src/infrastructure/pdf_parser.cpp)
target_include_directories(interview_infrastructure PUBLIC src)
target_link_libraries(interview_infrastructure PUBLIC interview_domain PRIVATE interview_json CURL::libcurl PkgConfig::POPPLER)
add_library(interview_storage src/infrastructure/storage/report_store.cpp src/infrastructure/storage/redis_cache.cpp src/infrastructure/storage/identity.cpp)
target_link_libraries(interview_storage PUBLIC interview_domain PRIVATE interview_database_bootstrap PkgConfig::MYSQL PkgConfig::HIREDIS OpenSSL::Crypto)
add_library(interview_accounts src/accounts/service.cpp src/accounts/crypto.cpp src/accounts/mysql_users.cpp src/accounts/redis_sessions.cpp)
target_include_directories(interview_accounts PUBLIC src)
target_link_libraries(interview_accounts PUBLIC interview_domain PRIVATE interview_database_bootstrap PkgConfig::MYSQL PkgConfig::HIREDIS OpenSSL::Crypto)
add_executable(interview_server src/api/main.cpp src/api/service.cpp src/api/start_interview.cpp src/api/account.cpp)
# 聚合构建时维持 build/server/interview_server 的启动路径。
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 set_target_properties(interview_server PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/..")
endif()
target_link_libraries(interview_server PRIVATE interview_protocol interview_application interview_infrastructure interview_storage interview_accounts)
include(CTest)
if(BUILD_TESTING)
 add_executable(accounts_test tests/accounts_test.cpp)
 target_link_libraries(accounts_test PRIVATE interview_accounts)
 add_test(NAME backend_accounts COMMAND accounts_test)
 add_executable(question_stream_test tests/question_stream_test.cpp)
 target_link_libraries(question_stream_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME question_stream_preview COMMAND question_stream_test)
 add_executable(report_storage_test tests/report_storage_test.cpp)
 target_link_libraries(report_storage_test PRIVATE interview_storage interview_application interview_infrastructure)
 add_executable(report_failure_test tests/report_failure_test.cpp)
 target_link_libraries(report_failure_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME report_persistence_failure_rollback COMMAND report_failure_test)
 add_executable(model_output_test tests/model_output_test.cpp)
 target_link_libraries(model_output_test PRIVATE interview_infrastructure interview_json)
 add_test(NAME model_output_stream COMMAND model_output_test)
 add_executable(pdf_test tests/pdf_test.cpp)
 target_link_libraries(pdf_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME pdf_validation COMMAND pdf_test)
 add_executable(engine_test tests/engine_test.cpp)
 target_link_libraries(engine_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME engine_state_machine COMMAND engine_test)
 add_executable(rpc_smoke tests/rpc_smoke.cpp)
 target_link_libraries(rpc_smoke PRIVATE interview_protocol)
endif()
