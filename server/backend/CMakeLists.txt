cmake_minimum_required(VERSION 3.24)
project(aiInterviewBackend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)

target_link_libraries(interview_json PUBLIC interview_domain)

target_include_directories(interview_connection_pools PUBLIC src)
target_link_libraries(interview_connection_pools PUBLIC interview_domain PkgConfig::MYSQL PkgConfig::HIREDIS)
add_library(interview_storage src/infrastructure/storage/report_store.cpp src/infrastructure/storage/redis_cache.cpp src/infrastructure/storage/identity.cpp)
target_link_libraries(interview_storage PUBLIC interview_domain PRIVATE interview_connection_pools interview_database_bootstrap PkgConfig::MYSQL PkgConfig::HIREDIS OpenSSL::Crypto)
add_library(interview_accounts src/accounts/service.cpp src/accounts/crypto.cpp src/accounts/mysql_users.cpp src/accounts/redis_sessions.cpp)
target_include_directories(interview_accounts PUBLIC src)
target_link_libraries(interview_accounts PUBLIC interview_domain PRIVATE interview_connection_pools interview_database_bootstrap PkgConfig::MYSQL PkgConfig::HIREDIS OpenSSL::Crypto)
add_library(interview_rpc src/api/native.cpp src/api/service.cpp src/api/start_interview.cpp src/api/account.cpp)
target_link_libraries(interview_rpc PUBLIC interview_protocol interview_application interview_infrastructure interview_storage interview_accounts)
add_library(interview_web src/transport/web/protocol.cpp src/transport/web/server.cpp src/transport/web/http.cpp src/transport/web/routes.cpp src/transport/web/accounts.cpp src/transport/web/rpc.cpp src/transport/web/static_files.cpp)
target_link_libraries(interview_web PUBLIC interview_rpc Boost::headers Threads::Threads)
add_executable(interview_server src/api/main.cpp)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 set_target_properties(interview_server PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/..")
endif()
target_link_libraries(interview_server PRIVATE interview_web interview_discovery)
