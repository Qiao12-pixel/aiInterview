cmake_minimum_required(VERSION 3.24)
project(aiInterviewGateway LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../proto" "${CMAKE_CURRENT_BINARY_DIR}/protocol")
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_library(gateway_core
 src/config.cpp src/common/codec.cpp src/common/random.cpp
 src/transport/grpc_client.cpp src/transport/sse.cpp src/transport/http.cpp src/transport/websocket.cpp
 src/routes/accounts.cpp src/routes/interviews.cpp src/routes/reports.cpp src/routes/static_files.cpp)
target_include_directories(gateway_core PUBLIC src)
target_compile_definitions(gateway_core PRIVATE GATEWAY_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(gateway_core PUBLIC interview_protocol nlohmann_json::nlohmann_json Boost::headers Threads::Threads OpenSSL::Crypto)
add_executable(interview_gateway src/main.cpp src/transport/server.cpp)
target_link_libraries(interview_gateway PRIVATE gateway_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(gateway_unit_tests tests/unit.cpp)
 target_link_libraries(gateway_unit_tests PRIVATE gateway_core)
 add_test(NAME gateway_unit COMMAND gateway_unit_tests)
endif()
