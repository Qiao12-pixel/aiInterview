cmake_minimum_required(VERSION 3.24)
project(aiInterviewServer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../proto" "${CMAKE_CURRENT_BINARY_DIR}/protocol")
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-cpp)
# 应用层只依赖领域与端口；具体 HTTP/PDF 实现不反向进入业务库。
add_library(interview_domain INTERFACE)
target_include_directories(interview_domain INTERFACE include)
target_link_libraries(interview_domain INTERFACE nlohmann_json::nlohmann_json)
add_library(interview_json src/common/json_output.cpp src/common/base64.cpp)
target_link_libraries(interview_json PUBLIC interview_domain)
add_library(interview_application src/application/engine.cpp src/application/answer.cpp)
target_link_libraries(interview_application PUBLIC interview_domain interview_json)
add_library(interview_infrastructure src/infrastructure/llm.cpp src/infrastructure/llm/contract.cpp src/infrastructure/llm/mock.cpp src/infrastructure/llm/config.cpp src/infrastructure/llm/http.cpp src/infrastructure/llm/stream.cpp src/infrastructure/pdf_parser.cpp)
target_include_directories(interview_infrastructure PUBLIC include)
target_link_libraries(interview_infrastructure PUBLIC interview_domain PRIVATE interview_json CURL::libcurl PkgConfig::POPPLER)
add_executable(interview_server src/api/main.cpp src/api/service.cpp)
target_link_libraries(interview_server PRIVATE interview_protocol interview_application interview_infrastructure)
include(CTest)
if(BUILD_TESTING)
 add_executable(model_output_test tests/model_output_test.cpp)
 target_link_libraries(model_output_test PRIVATE interview_infrastructure interview_json)
 add_test(NAME model_output_stream COMMAND model_output_test)
 add_executable(pdf_test tests/pdf_test.cpp)
 target_link_libraries(pdf_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME pdf_validation COMMAND pdf_test)
 add_executable(engine_test tests/engine_test.cpp)
 target_link_libraries(engine_test PRIVATE interview_application interview_infrastructure)
 add_test(NAME engine_state_machine COMMAND engine_test)
 add_executable(rpc_smoke tests/rpc_smoke.cpp)
 target_link_libraries(rpc_smoke PRIVATE interview_protocol)
endif()
